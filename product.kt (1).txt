@Composable
fun Greeting (modifier: Modifier = Modifier)
{

    val productViewModel= remember {ProductViewModel()}
    val navController = rememberNavController()
    NavHost(navController = navController, startDestination = "first")
    {
        composable(route = "first"){
            FirstScreen(
               productViewModel,
                navController
            )
        }
        composable(route = "second"){
            SecondScreen(
                productViewModel,
                navController
            )
        }
        composable(route = "third"){ThirdScreen(productViewModel,navController)}
        composable(route = "fourth"){FourthScreen(productViewModel, navController)}
        composable(route = "fifth"){FifthScreen(productViewModel, navController)}
    }

}


class ProductViewModel: ViewModel()
{
    var product by mutableStateOf("")
    var price by mutableStateOf("")
    var quantity by mutableStateOf("")
    var cost by mutableStateOf("")
    var tax by mutableStateOf("")
    var discount by mutableStateOf("")
    var totalcost by mutableStateOf("")
    var netcost by mutableStateOf("")
}

{
    Column(

                text="PRODUCT",

        Row(
            horizontalArrangement=Arrangement.SpaceEvenly,
            verticalAlignment=Alignment.CenterVertically,
            modifier = Modifier.fillMaxWidth()
        )
        {
            Text(
                text="PRICE",
                fontSize = 30.sp

            )

            TextField(
                value=productViewModel.price,

            )
        }


        Row(
            horizontalArrangement=Arrangement.SpaceEvenly,

            modifier = Modifier.fillMaxWidth()

@Composable
fun FourthScreen(productViewModel: ProductViewModel, navController: NavController)
{
    //no manual data entry
    //declare variables using by remember
    var totalcostvalue by remember { mutableStateOf("") }
    var netcostvalue by remember { mutableStateOf("") }
    // read cost value, discount and tax value from view model
    var costDouble= productViewModel.cost.toDoubleOrNull()?:0.0
    var taxDouble = productViewModel.tax.toDoubleOrNull()?:0.0
    var discountDouble = productViewModel.discount.toDoubleOrNull()?:0.0

    //use the value of cost to calculate tax and discount
    var totalcostDouble= costDouble+taxDouble
    var netcostDouble = totalcostDouble-discountDouble
    //display the total and netcosts in their text fields
    totalcostvalue = totalcostDouble.toString()
    netcostvalue=netcostDouble.toString()
    //update the tax and discount properties
    productViewModel.totalcost=totalcostvalue
    productViewModel.netcost=netcostvalue

    Column(
        verticalArrangement=Arrangement.SpaceEvenly,
        horizontalAlignment=Alignment.CenterHorizontally,
        modifier = Modifier.fillMaxSize()
    )
    {
        Text(
            text="FOURTH SCREEN",
            fontSize = 40.sp
        )

        Row(
            horizontalArrangement=Arrangement.SpaceEvenly,
            verticalAlignment=Alignment.CenterVertically,
            modifier = Modifier.fillMaxWidth()
        )
        {
            Text(
                text="TOTAL COST",
                fontSize = 30.sp
            )

            TextField(
                value=totalcostvalue,
                onValueChange={totalcostvalue=it}
            )
        }


        Row(

        }


        Row(

}
