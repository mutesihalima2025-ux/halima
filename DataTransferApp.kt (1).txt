@Composable
fun Greeting(modifier: Modifier = Modifier)
{
        val navController = rememberNavController()
        val studentviewmodel = remember { StudentViewModel() }

        NavHost(navController = navController, startDestination = "first")
        {
            composable(route = "first"){FirstScreen(studentviewmodel, navController)}
            composable(route = "second"){SecondScreen(studentviewmodel, navController)}
            composable(route = "third"){ThirdScreen(studentviewmodel,navController)}
            composable(route = "fourth"){FourthScreen(studentviewmodel, navController)}
            composable(route = "fifth"){FifthScreen(studentviewmodel, navController)}
        }

}

    var fees by  mutableStateOf("")
    var phone by  mutableStateOf("")
    var email by  mutableStateOf("")
}
